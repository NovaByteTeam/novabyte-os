  function extOf(name) {

  // Cheap pre-check before attempting Uint8Array.fromBase64. Avoids
  // throwing inside fromBase64 for plain-text content.
  function isLikelyBase64(str) {
    return (
      str.length > 0 &&
      str.length % 4 === 0 &&
      /^[A-Za-z0-9+/]+={0,2}$/.test(str)
    );
  }
 
  // Coerces whatever shape track.content happens to be into a typed
  // array we can wrap in a Blob. Returns null for unsupported shapes
  // or empty payloads.
  function normalizeBuffer(raw) {
    if (!raw) return null;
    if (raw instanceof ArrayBuffer || ArrayBuffer.isView(raw)) return raw;
 
    if (typeof raw === 'string') {
      // Prefer native base64 decoding when the string looks like base64;
      // otherwise treat it as UTF-8 text. Native decoding is constant-
      // time and avoids the per-byte charCodeAt loop the original used.
      if (isLikelyBase64(raw)) {
        try {
          return Uint8Array.fromBase64(raw);
        } catch {
          // fall through and treat as UTF-8 text
        }
      }
      return new TextEncoder().encode(raw);
    }
 
    if (typeof raw === 'object') {
      // Legacy shape: array-like object with numeric keys.
      const keys = Object.keys(raw);
      const out = new Uint8Array(keys.length);
      for (let i = 0; i < keys.length; i++) out[i] = raw[i] ?? 0;
      return out;
    }
 
    return null;
  }
 
  function loadPrefs(key) {

    // the localStorage key is adopted once + kept as the fail-soft fallback.
    if (typeof window !== 'undefined' && window.NovaAppConfig) {
      try {
        const value = window.NovaAppConfig.get('com.nbosp.music', 'prefs', { legacyKey: key, fallback: null });
        return value && typeof value === 'object' && !Array.isArray(value) ? value : {};
      } catch (_) { /* fall through to legacy */ }
    }
    try {
      const raw = localStorage.getItem(key);
      return raw ? JSON.parse(raw) : {};
    } catch (err) {
      // Corrupt JSON in storage is recoverable — start fresh.
      console.warn('[Music] Discarding unreadable prefs:', err);
      return {};
    }
  }
 
  function savePrefs(key, prefs) {
    if (typeof window !== 'undefined' && window.NovaAppConfig) {
      try {
        window.NovaAppConfig.set('com.nbosp.music', 'prefs', prefs, { legacyKey: key });
        return;
      } catch (_) { /* fall through to legacy */ }
    }

      lsSave(key, prefs);
    } catch (err) {
      // Quota / privacy mode / disabled storage — non-fatal, but log
      // so the user has a clue why their prefs "forgot" themselves.
      console.warn('[Music] Could not persist prefs:', err);
    }
  }
 
  // Formats seconds as m:ss or h:mm:ss. Falls back to 0:00 for NaN /
  // Infinity (which <audio> emits before metadata has loaded).
  function fmtTime(seconds) {

    const h = Math.floor(total / 3600);
    const m = Math.floor((total % 3600) / 60);

    const ss = String(s).padStart(2, '0');
    return h > 0 ? `${h}:${String(m).padStart(2, '0')}:${ss}` : `${m}:${ss}`;
  }

      // Wrap audio.play() so rejections are reported consistently.

            title: _t('apps.music.notify.title'),

      async function getUrl(track) {
        if (!track) return null;
        const cached = blobCache.get(track.id);
        if (cached) return cached;
 
        let data;
        try {
          data = normalizeBuffer(track.content);
          if (!data && nativeFs && typeof track.id === 'string') {
            data = await nativeFs.readFile(track.id);
          }
        } catch (err) {
          console.warn('[Music] Could not read track content:', err);
          return null;
        }
        if (!data || data.byteLength === 0) return null;
 
        // Prefer a stored audio/* MIME; otherwise fall back to the
        // extension-derived one. A wrong MIME on the blob causes
        // NS_ERROR_DOM_MEDIA_METADATA_ERR in Firefox even when the
        // bytes themselves are valid.
        const storedMime = track.mimeType || '';
        const mime = storedMime.startsWith('audio/') ? storedMime : mimeFromName(track.name);
 
        try {
          const blob = new Blob([data], mime ? { type: mime } : undefined);
          const url = URL.createObjectURL(blob);
          blobCache.set(track.id, url);
          return url;
        } catch (err) {
          console.warn('[Music] Could not create object URL:', err);
          return null;
        }
      }
 
      function buildQueue(startId) {
        queue = [...tracks];
        if (shuffle) {
          // Fisher–Yates — unbiased shuffle.
          for (let i = queue.length - 1; i > 0; i--) {
            const j = Math.floor(Math.random() * (i + 1));
            [queue[i], queue[j]] = [queue[j], queue[i]];
          }
        }
        if (startId != null) {
          const si = queue.findIndex((t) => t.id === startId);
          if (si > -1) queueIdx = si;
        }
      }
 
      function playIdx(i) {
        if (!queue.length) return;
        queueIdx = ((i % queue.length) + queue.length) % queue.length;
        playTrack(queue[queueIdx]);
      }

      // button, scrub bar and row suffixes all settle on "paused at start".

      // ---- Library scan & render ----
 
      function isAudioFile(file) {
        return file?.type === 'file' && (
          file.mimeType?.startsWith('audio/') || AUDIO_EXTENSIONS.has(extOf(file.name))
        );
      }
 
      let libraryScanVersion = 0;
      function queryAllMedia() {
        const found = [];
        for (let offset = 0; offset < 5000; offset += 500) {
          const page = window.NovaByteMediaStore.query({ type: 'audio', limit: 500, offset });
          found.push(...page);
          if (page.length < 500) break;
        }
        return found;
      }
 
      async function scanLibrary() {
        const scanVersion = ++libraryScanVersion;
        try {
          if (!window.NovaByteMediaStore?.query) throw new Error('NovaByte MediaStore is unavailable');
          const found = queryAllMedia()
            .map(item => FS.files.get(item.id) || {
              ...item,
              type: 'file',
              content: null,
              parentId: null,
            });
 
          // A file opened from Files is a user-picked item and may live outside
          // Music; include only that explicit intent in addition to the library.
          const openedFile = openedFileId ? FS?.files?.get(openedFileId) : null;
          if (isAudioFile(openedFile) && !found.some(file => file.id === openedFile.id)) {
            found.push(openedFile);
          }
          found.sort((a, b) => (a.name || '').localeCompare(b.name || ''));
 
          if (scanVersion !== libraryScanVersion) return;
          tracks.length = 0;
          for (const track of found) tracks.push(track);
        } catch (err) {
          console.error('[Music] Library scan failed:', err);
          Notify.show({ title: _t('apps.music.notify.title'), body: _t('apps.music.notify.couldNotReadFolder', { message: err.message }), type: 'error', appName: 'Music' });
        }
 
        // Revoke object URLs for tracks that have disappeared — without
        // this the blobCache leaks one object URL per ever-loaded track.
        const validIds = new Set(tracks.map((t) => t.id));
        for (const [id, url] of blobCache) {
          if (!validIds.has(id)) {
            URL.revokeObjectURL(url);
            blobCache.delete(id);
          }
        }
 
        const n = tracks.length;
        libCount.textContent = n ? _t('apps.music.library.trackCount', { count: n, s: n > 1 ? 's' : '' }) : '';
        // Bugfix: every fs event re-ran scanLibrary, which rebuilt AND
        // re-shuffled the queue mid-playback — the "up next" order kept
        // changing under the user. Only rebuild (and re-shuffle) when the
        // player is idle; while a track is playing, keep the queue order the
        // user is listening through. The cursor still points at the current
        // track; a vanished track is skipped on the next manual interaction.
        if (!(!audio.paused && audio.src)) buildQueue(queue[queueIdx]?.id);
        renderList();
 
        if (!initialFileHandled && openedFileId) {
          initialFileHandled = true;
          // Bugfix: openedIdx was computed against the sorted `tracks` array
          // but played from the (possibly shuffled) `queue`, so with shuffle
          // on the WRONG track could autoplay. Look the id up IN the queue;
          // fall back to the old tracks-index behavior if it is absent.
          const openedIdx = queue.findIndex(track => track.id === openedFileId);
          if (openedIdx !== -1) {
            playIdx(openedIdx);
          } else {
            const trackIdx = tracks.findIndex(track => track.id === openedFileId);
            if (trackIdx !== -1) playIdx(trackIdx);
          }
        }
      }
