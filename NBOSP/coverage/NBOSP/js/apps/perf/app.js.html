    if (typeof _origFetch === 'function') {
      window.fetch = function (...args) {
        const start = performance.now();

        const method = (args[1]?.method || 'GET').toUpperCase();

          const contentType = res.headers.get('content-type') || '';

          const finish = (body, bodySkipped) => pushNetEntry({
            source: 'os', method, url, status: res.status,
            duration: Math.round(performance.now() - start),
            error: null, label, time: Date.now(),

          });

            // branch's body at all, but calling clone() unconditionally
            // before the type check would make every response (including
            // huge binary ones) briefly double-buffer for no reason.
            finish(null, true);
            return res;
          }

          // the calling app still needs to read itself — .text() on the
          // original response would leave nothing for the app's own
          // .then(res => res.json()) etc.

          }).catch(() => finish(null, true));
          return res;
        }).catch(err => {

        });

    XMLHttpRequest.prototype.send = function (...args) {
      const start = performance.now();
      const label = labelForFetch();

        const isTextish = BODY_CAPTURE_TEXT_TYPES.test(contentType);
        // responseText throws if responseType is e.g. 'blob' or
        // 'arraybuffer' — those are exactly the cases isTextish should
        // already exclude via content-type, but an app could still set a
        // binary responseType against a server that mislabels its
        // content-type, so this stays defensive rather than trusting the
        // header alone.

        if (isTextish) {
          try {
            const text = this.responseText;
            bodySkipped = false;
            body = text.length > BODY_CAPTURE_MAX_BYTES ? _t('apps.perf.bodyTruncated', { text: text.slice(0, BODY_CAPTURE_MAX_BYTES) }) : text;
          } catch { /* non-text responseType — leave bodySkipped true */ }
        }
        pushNetEntry({
          source: 'os',
          method: this._nbPerfMethod,
          url: this._nbPerfUrl,
          status: this.status || null,
          duration: Math.round(performance.now() - start),
          error: this.status === 0 ? _t('apps.perf.networkError') : null,
          label,
          time: Date.now(),
          body,
          bodySkipped,
          contentType,
        });
      });
      return _origXHRSend.apply(this, args);
    };

    // Attribution shape differs by entry type and is NOT normalized into a

    // NOTE on window attribution: every app window here is a <div> in one
    // shared document (confirmed — no iframes anywhere in this codebase),
    // so the browser's own long-task attribution (LoAF's `invoker`, legacy
    // `attribution[].containerType`) has no frame boundary to attribute
    // *to* and will just say "self" or be empty. True per-window
    // attribution isn't something this API can give us here. What we
    // record instead is which window was focused at the moment the task
    // fired — a correlation, not a proof of causation, and labeled as such
    // in the UI rather than presented like a real culprit.
    function pushLongTaskEntry(e) {
      const entry = {
        time: e.startTime,
        duration: Math.round(e.duration),
        type: longTaskEntryType,
        detail: null,
        focusedWindow: (typeof OS !== 'undefined' && OS.windows && OS.focusedWindowId)
          ? (OS.windows.get(OS.focusedWindowId)?.appId || null)
          : null,
      };

        // Attribute to the single longest-running script in the frame —
        // usually the actual culprit rather than an innocent bystander
        // swept up in the same frame.
        const top = [...e.scripts].sort((a, b) => (b.duration || 0) - (a.duration || 0))[0];
        entry.detail = {
          source: top.sourceFunctionName || top.sourceURL || _t('apps.perf.unknownScript'),
          invoker: top.invoker || null,
          scriptDuration: Math.round(top.duration || 0),
          scriptCount: e.scripts.length,
        };

        const a = e.attribution[0];
        entry.detail = {
          source: a.containerName || a.containerSrc || a.containerType || _t('apps.perf.unknownContainer'),
          invoker: null,
          scriptDuration: null,
          scriptCount: e.attribution.length,
        };
      }

    function measureFrame() {
      const now = performance.now();

        // Freeze the counters instead of letting frameCount and elapsed

        // producing a garbage spike on the chart.

        return;

        memHistory.push(mb);
        memHistory.shift();

    function themeColor(token, fallback) {

    function drawChart(cv, history, color, referenceY) {
      const ctx = cv.getContext('2d');
      const w = cv.width = cv.offsetWidth * 2;
      const h = cv.height = cv.offsetHeight * 2;
      ctx.scale(2, 2);
      const cw = w / 2, ch = h / 2;
      ctx.clearRect(0, 0, cw, ch);
 
      const max = Math.max(referenceY || 0, 1, ...history);
      const stepX = cw / (history.length - 1);
 
      if (referenceY) {
        const yRef = ch - (referenceY / max) * ch;
        ctx.globalAlpha = 0.2;
        ctx.strokeStyle = themeColor('--text-success', '#00ff00');
        ctx.lineWidth = 1;
        ctx.beginPath();
        ctx.moveTo(0, yRef);
        ctx.lineTo(cw, yRef);
        ctx.stroke();
        ctx.globalAlpha = 1;
      }
 
      ctx.strokeStyle = color;
      ctx.lineWidth = 1.5;
      ctx.beginPath();
      history.forEach((v, i) => {
        const x = i * stepX;
        const y = ch - (v / max) * ch;
        if (i === 0) ctx.moveTo(x, y); else ctx.lineTo(x, y);
      });
      ctx.stroke();
 
      ctx.lineTo(cw, ch);
      ctx.lineTo(0, ch);
      ctx.closePath();
      ctx.globalAlpha = 0.08;
      ctx.fillStyle = color;
      ctx.fill();
      ctx.globalAlpha = 1;
    }
 
    function updateStats(domCount) {

      // Sustained-low-FPS alert: require several consecutive sub-threshold
      // samples before flagging, so one dip during a window-open animation
      // doesn't falsely trip it. Resets the streak on any healthy sample.
