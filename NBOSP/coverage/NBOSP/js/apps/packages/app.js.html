                fileBox.appendChild(createEl('div', { textContent: path, style: 'font-size:11px;color:var(--text-muted);font-family:monospace;margin-bottom:2px;' }));
                let text;
                try {
                  text = new TextDecoder('utf-8', { fatal: true }).decode(bytes);
                } catch {
                  text = _t('apps.packages.binaryContentNote', { count: bytes.length });
                }
                const MAX_PREVIEW = 5000;
                const truncated = text.length > MAX_PREVIEW;
                fileBox.appendChild(createEl('pre', {
                  textContent: truncated ? text.slice(0, MAX_PREVIEW) + '\n' + _t('apps.packages.truncatedSuffix') : text,
                  style: 'margin:0;padding:8px;background:var(--bg-elevated);border:1px solid var(--border-subtle);border-radius:4px;font-size:10px;white-space:pre-wrap;word-break:break-all;max-height:140px;overflow:auto;'
                }));
                filesPreview.appendChild(fileBox);
              });
            } catch (e) {
              filesPreview.appendChild(createEl('div', { textContent: _t('apps.packages.errExtractFiles', { message: e.message }), style: 'font-size:11px;color:var(--text-danger);' }));
            }
          }
 
          lastVerifiedPackage = pkg; // handed to Install tab below
        } catch (e) {
          showResult(box, false, { error: e.message });
        }
      }, { signal: ac.signal });
 
      panel.appendChild(verifyBtn);
      panel.appendChild(box);
      panel.appendChild(exportButtons(() => lastVerifiedPackage, 'verified'));
 
      const filesPreview = createEl('div', { style: 'margin-top:12px;' });
      panel.appendChild(filesPreview);
    })();
 
    // ── Install tab ──────────────────────────────────────────────────────
    // AppPackage.installPackage/uninstallPackage exist and are exported
    // (confirmed in app-package.js) but had no UI anywhere in this file —
    // Sign and Verify could produce and validate a package but never
    // actually install one, which is the actual point of a packaging tool.
    (function buildInstallTab() {
      const panel = panels.install;
      panel.appendChild(createEl('div', {
        textContent: _t('apps.packages.installTabNote'),
        style: 'font-size:11px;color:var(--text-muted);margin-bottom:10px;line-height:1.4;'
      }));
 
      const sourceRow = createEl('div', { style: 'display:flex;gap:8px;margin-bottom:10px;' });
      const useSignedBtn = createEl('button', { textContent: _t('apps.packages.useSignedBtn'), style: 'padding:5px 10px;background:transparent;color:var(--text-muted);border:1px solid var(--border-subtle);border-radius:4px;cursor:pointer;font-size:11px;' });
      const useVerifiedBtn = createEl('button', { textContent: _t('apps.packages.useVerifiedBtn'), style: 'padding:5px 10px;background:transparent;color:var(--text-muted);border:1px solid var(--border-subtle);border-radius:4px;cursor:pointer;font-size:11px;' });
      sourceRow.appendChild(useSignedBtn);
      sourceRow.appendChild(useVerifiedBtn);
      panel.appendChild(sourceRow);
 
      panel.appendChild(createEl('label', { textContent: _t('apps.packages.packageJsonLabel'), style: 'display:block;font-size:11px;color:var(--text-muted);margin-bottom:4px;' }));
      const textarea = createEl('textarea', {
        placeholder: _t('apps.packages.installPlaceholder'),
        style: 'width:100%;box-sizing:border-box;height:140px;padding:8px;background:var(--bg-elevated);border:1px solid var(--border-subtle);border-radius:4px;color:var(--text-primary);font-family:monospace;font-size:11px;resize:vertical;'
      });
      panel.appendChild(textarea);
 
      useSignedBtn.addEventListener('click', () => {
        if (!lastSignedPackage) return;
        textarea.value = JSON.stringify(lastSignedPackage, null, 2);
      }, { signal: ac.signal });
      useVerifiedBtn.addEventListener('click', () => {
        if (!lastVerifiedPackage) return;
        textarea.value = JSON.stringify(lastVerifiedPackage, null, 2);
      }, { signal: ac.signal });
 
      // NOTE: installPackage() no longer takes a typed "verification key" —
      // verification now checks the package's signature against every
      // public key in TrustStore (the same list NBOSP itself ships), so a
      // package either matches a trusted signer or it doesn't. There's
      // nothing to type in here.
      const optRow = createEl('div', { style: 'display:flex;flex-direction:column;gap:4px;margin-bottom:10px;font-size:11px;' });
      const skipVerifyLabel = createEl('label', { style: 'display:flex;align-items:center;gap:6px;cursor:pointer;' });
      const skipVerifyCb = createEl('input', { type: 'checkbox' });
      skipVerifyLabel.appendChild(skipVerifyCb);
      skipVerifyLabel.appendChild(createEl('span', { textContent: _t('apps.packages.skipVerifyLabel') }));
      optRow.appendChild(skipVerifyLabel);
 
      const allowUnverifiedLabel = createEl('label', { style: 'display:flex;align-items:center;gap:6px;cursor:pointer;' });
      const allowUnverifiedCb = createEl('input', { type: 'checkbox' });
      allowUnverifiedLabel.appendChild(allowUnverifiedCb);
      allowUnverifiedLabel.appendChild(createEl('span', { textContent: _t('apps.packages.allowUnverifiedLabel') }));
      optRow.appendChild(allowUnverifiedLabel);
 
      const forceLabel = createEl('label', { style: 'display:flex;align-items:center;gap:6px;cursor:pointer;' });
      const forceCb = createEl('input', { type: 'checkbox' });
      forceLabel.appendChild(forceCb);
      forceLabel.appendChild(createEl('span', { textContent: _t('apps.packages.forceOverwriteLabel') }));
      optRow.appendChild(forceLabel);
      panel.appendChild(optRow);
 
      const installBtn = createEl('button', { textContent: _t('apps.packages.installPackageBtn'), style: 'padding:7px 16px;background:var(--accent);color:var(--text-on-accent);border:none;border-radius:4px;cursor:pointer;font-size:12px;' });
      const box = resultBox();
 
      installBtn.addEventListener('click', async () => {
        const raw = textarea.value.trim();
        if (!raw) {
          showResult(box, false, { error: _t('apps.packages.errPasteJsonOrButton') });
          return;
        }
        let pkg;
        try {
          pkg = JSON.parse(raw);
        } catch {
          showResult(box, false, { error: _t('apps.packages.errInvalidJson') });
          return;
        }
 
        // AppRegistry is referenced with optional chaining inside
        // app-package.js itself (AppRegistry?.registerApp(...)), meaning
        // even that module's own author didn't assume it's always present.
        // Check explicitly here too, rather than letting installPackage's
        // internal `?.` silently produce a "success" with `app: undefined`.
        if (typeof AppRegistry === 'undefined' || !AppRegistry) {
          showResult(box, false, { error: _t('apps.packages.errNoAppRegistry') });
          return;
        }
 
        installBtn.disabled = true;
        installBtn.textContent = _t('apps.packages.installingBtn');
        try {
          const options = {
            skipVerify: skipVerifyCb.checked,
            trustStore: (typeof TrustStore !== 'undefined' && TrustStore.list) ? TrustStore.list() : [],
            revocationCheck: (typeof TrustStore !== 'undefined' && TrustStore.isRevoked) ? TrustStore.isRevoked : undefined,
            allowUnverified: allowUnverifiedCb.checked,
            allowIntegrityFallback: allowUnverifiedCb.checked,
            force: forceCb.checked,
            source: 'packages-app',
          };
          const result = await AppPackage.installPackage(pkg, options);
          if (!result?.success || !result.app) {
            showResult(box, false, { error: _t('apps.packages.errInstallFailed', { error: result?.error || _t('apps.packages.errorUnknown') }) });
            return;
          }
 
          const PackageStore = window.NovaAppPackageStore || null;
          const appData = {
            id: pkg.manifest.id,
            name: pkg.manifest.name,
            version: pkg.manifest.version,
            entry: pkg.manifest.entry,
            icon: pkg.manifest.icon,
            description: pkg.manifest.description || '',
            category: pkg.manifest.category || 'other',
            permissions: pkg.manifest.permissions || [],
            optionalPermissions: pkg.manifest.optionalPermissions || [],
            allowNestedWebview: pkg.manifest.allowNestedWebview === true,
            manifest: pkg.manifest,
            files: pkg.files || {},
            signature: pkg.signature,
            verified: result.verified,
            signer: result.signer,
            source: 'packages-app',
            installedDate: new Date().toISOString(),
          };
 
          if (PackageStore?.installApp) {
            try {
              const meta = await PackageStore.installApp(appData);
              const list = PackageStore.loadRegistry ? PackageStore.loadRegistry() : [];
              const idx = list.findIndex(a => a.id === appData.id);
              const entry = { ...(idx > -1 ? list[idx] : {}), ...meta, ...appData, id: appData.id };
              if (idx > -1) list[idx] = entry; else list.push(entry);
              PackageStore.saveRegistry(list);
            } catch (e) {
              console.warn('[Packages] Failed to persist installed app files:', e);
            }
          }
 
          const registered = result.app || OS.apps[appData.id];
          const resolvedIcon = appData.icon || registered?.icon || '/assets/no_app_icon.svg';
 
          if (appData.icon && !/^data:|^https?:\/\//i.test(appData.icon) && appData.files[appData.icon]) {
            const encoded = appData.files[appData.icon];
            const ext = (appData.icon.split('.').pop() || '').toLowerCase();
            const mime = ext === 'svg' ? 'image/svg+xml'
              : ext === 'jpg' || ext === 'jpeg' ? 'image/jpeg'
              : ext === 'gif' ? 'image/gif'
              : ext === 'webp' ? 'image/webp'
              : ext === 'ico' ? 'image/x-icon'
              : 'image/png';
            appData.icon = `data:${mime};base64,${encoded}`;
          }
 
          const cfg = {
            id: appData.id,
            name: appData.name,
            icon: appData.icon,
            description: appData.description,
            version: appData.version,
            entry: appData.entry,
            defaultSize: [800, 600],
            minSize: [400, 300],
            maxSize: null,
            resizable: true,
            frame: true,
            sandbox: { allowSameOrigin: false, allowScripts: true, allowForms: true, allowPopups: false },
            categories: [appData.category || 'other'],
            permissions: appData.permissions,
            optionalPermissions: appData.optionalPermissions,
            allowNestedWebview: appData.allowNestedWebview,
            manifest: pkg.manifest,
            // See the matching comment in nova-app-launcher.js's buildNovaAppConfig —
            // same missing field, this install path builds its own separate
            // config object instead of sharing that one.
            handles: appData.handles ?? appData.manifest?.handles ?? undefined,
            init: (content, state, options) => AppRegistry.launchApp(appData.id, content, state, options),
          };
          OS.apps[appData.id] = cfg;
          const ri = APP_REGISTRY.findIndex(a => a.id === appData.id);
          if (ri > -1) APP_REGISTRY[ri] = cfg; else APP_REGISTRY.push(cfg);
 
          showResult(box, true, { ...result, persisted: true, message: _t('apps.packages.installedRegisteredMsg', { app: appData.name, version: appData.version }) });
          if (typeof renderLaunchpad === 'function') renderLaunchpad();
          if (typeof WM !== 'undefined' && WM.updateTaskbar) WM.updateTaskbar();
          if (typeof renderDesktopIcons === 'function') renderDesktopIcons();
          Notify.show({ title: _t('apps.packages.appInstalledTitle'), body: _t('apps.packages.appInstalledBody', { app: appData.name, version: appData.version }), type: 'success', appName: 'Packages' });
        } catch (e) {
          showResult(box, false, { error: e.message });
        } finally {
          installBtn.disabled = false;
          installBtn.textContent = _t('apps.packages.installPackageBtn');
        }
      }, { signal: ac.signal });
 
      panel.appendChild(installBtn);
      panel.appendChild(box);
    })();
 
    // Activate the first tab only now — both panels are built and attached
    // to the live DOM at this point. Doing this during the TABS.forEach
    // loop above (the original approach) fired the click before
    // buildSignTab() had added any content and before `scroll` was even
    // attached to `content`, which is what caused the reported bug: the
    // Sign tab showed blank on first launch and only rendered correctly
    // after switching tabs forced a fresh display toggle + reflow.
    tabBar.querySelector('button').click();
  }
