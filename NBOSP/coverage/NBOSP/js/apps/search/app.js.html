    // Direct egress to api.duckduckgo.com is blocked at this deployment's

    // Contract: POST /api/proxy with JSON { url, method, headers, body,

    // bodyEncoding } (body is a UTF-8 string for textual targets, base64

    function proxyCsrfToken() {
      try {
        // Multi-user: the CSRF token belongs to the active session, which is
        // the one this host fetches with (X-Nova-Session below) — the
        // boot-injected meta tag belongs to the anonymous page session.
        if (typeof NovaUserScope !== 'undefined' && NovaUserScope?.enabled?.() &&
            typeof window !== 'undefined' && window.SessionManager?.csrf) {
          const active = window.SessionManager.csrf();
          if (active) return active;
        }
        return document.querySelector('meta[name="csrf-token"]')?.content || '';
      } catch { return ''; }
    }
 
    function proxyFetch(url, signal) {
      const headers = { 'Content-Type': 'application/json' };
      const csrf = proxyCsrfToken();
      if (csrf) headers['X-CSRF-Token'] = csrf;
      try {
        if (typeof NovaUserScope !== 'undefined' && NovaUserScope?.enabled?.() &&
            typeof window !== 'undefined' && window.SessionManager?.token) {
          const token = window.SessionManager.token();
          if (token) headers['X-Nova-Session'] = token;
        }
      } catch { /* bearer is best-effort */ }
      return fetch('/api/proxy', {
        method: 'POST',
        credentials: 'same-origin',
        signal,
        headers,
        body: JSON.stringify({
          url,
          method: 'GET',
          headers: { 'Accept': 'application/json' },
          body: null,
          bodyEncoding: 'utf8',
        }),
      });
    }
 
    function proxyBodyText(envelope) {
      if (!envelope || typeof envelope.body !== 'string') return null;
      if (envelope.bodyEncoding === 'base64') {
        try {
          const bytes = Uint8Array.from(atob(envelope.body), ch => ch.charCodeAt(0));
          return new TextDecoder().decode(bytes);
        } catch { return null; }
      }
      return envelope.body;
    }
 
    async function fetchDDG(q, signal) {
      const res = await proxyFetch(ddgUrl(q), signal);
      if (!res.ok) throw new Error(`DDG proxy ${res.status}`);
      const envelope = await res.json();
      if (!envelope || typeof envelope.status !== 'number') {
        throw new Error('DDG proxy returned a malformed envelope');
      }
      if (envelope.status < 200 || envelope.status >= 300) {
        throw new Error(`DDG ${envelope.status}`);
      }
      const text = proxyBodyText(envelope);
      if (text === null) throw new Error('DDG proxy returned an unreadable body');
      return parseDDGResponse(JSON.parse(text));
    }

    async function doSearch(rawQuery) {

      const q  = rawQuery;
      const lq = q.trim().toLowerCase();

      let hasAnyLocal = false;

      const filesSec = buildSection(_t('apps.search.filesSection'), fileHits.slice(0, 12), (f) => {
        const path = filePathCache.get(f.id) || (f.name || '');
        const mime = fileMime(f);
        return buildRow(iconForMime(mime), f.name || _t('apps.search.unnamed'), path, () => {
          if (mime.startsWith('image/')) WM.createWindow('com.nbosp.gallery', { fileId: f.id });
          else if (mime.startsWith('audio/')) WM.createWindow('com.nbosp.music', { fileId: f.id });
          else if (typeof process !== 'undefined' && process.platform === 'linux'
              && FS.backend === 'linux-native') {
            WM.createWindow('com.nbosp.textedit', { path });
          } else {
            WM.createWindow('com.nbosp.textedit', { fileId: f.id });
          }
        });
      });
      if (filesSec) { results.appendChild(filesSec); hasAnyLocal = true; }
 
      const contactHits = localCache.searchContacts(lq);
      const contactsSec = buildSection(_t('apps.search.contactsSection'), contactHits, (c) =>
        buildRow('users', c.name || _t('apps.search.noName'), c.email || c.phone || '', () =>
          WM.createWindow('com.nbosp.contacts', { contactId: c.id })

      if (contactsSec) { results.appendChild(contactsSec); hasAnyLocal = true; }

          WM.createWindow('com.nbosp.downloads', { itemId: d.id })

      let webHits = getCachedWeb(q);

        loadingEl.textContent = _t('apps.search.fetchingWeb');
        results.appendChild(loadingEl);

          setCachedWeb(q, webHits);
        } catch (err) {
          if (err.name !== 'AbortError') console.warn('[nbosp-search] Web fetch failed:', err);
        }
 
        if (signal.aborted) return;
        loadingEl.remove();
      }
 
      if (webHits.length) {
        const webSec = buildSection(_t('apps.search.webSection'), webHits, (r) =>
          buildRow('globe', r.title, r.desc || r.href, () =>
            WM.createWindow('com.nbosp.browser', { url: r.href })
          )
        );

      } else {

          'globe',
          _t('apps.search.searchWebFor', { query: q }),
          _t('apps.search.openInBrowser'),
          () => WM.createWindow('com.nbosp.browser', {
            url: 'https://search.brave.com/search?' + new URLSearchParams({ q }),
          })
        );
        results.appendChild(fallback);
      }
 
      if (!hasAnyLocal && !webHits.length) showEmpty(q);
    }

    // INPUT HANDLING — debounce 80ms + independent prefetch
    // ════════════════════════════════════════════════════════════════════════
 
    let debounceId     = null;
    let lastPrefetchQ  = null;
 
    inp.addEventListener('input', () => {

      clearX.style.display = val ? '' : 'none';

      if (val.trim().length >= 3 && val !== lastPrefetchQ && !getCachedWeb(val)) {

          .catch(() => {});
      }
 
      debounceId = setTimeout(() => doSearch(val), 80);
    });
 
    clearX.addEventListener('click', () => {
      inp.value = '';
      clearX.style.display = 'none';
      fetchController?.abort();
      doSearch('');
      inp.focus();
    });

    // LIVE RE-LOCALISATION
    // doSearch(inp.value) re-derives the whole results surface (sections,
    // empty state, web fallback) from the current query; row interactions go
    // through delegation on `results`, so no listeners are re-attached. The
    // query text lives in inp.value and is passed straight back in, so
    // nothing the user typed is lost — only the labels change language.
    // ════════════════════════════════════════════════════════════════════════
    const localeHandler = () => {
      try {
        window.NovaI18n?.applyStaticDom?.(content);
        inp.placeholder = _t('apps.search.placeholder');
        inp.setAttribute('aria-label', _t('apps.search.searchLabel'));
        clearX.setAttribute('aria-label', _t('apps.search.clearSearchAria'));

      } catch (error) {
        console.warn('[Search] Language change re-render failed:', error);
      }
    };
    typeof OS !== 'undefined' && OS.events?.on?.('languageChanged', localeHandler);
    state.cleanups.push(() => typeof OS !== 'undefined' && OS.events?.off?.('languageChanged', localeHandler));
 
    doSearch('');
    inp.focus();
  },
